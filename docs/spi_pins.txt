=== MCU(U1) 上的 SPI 引脚 ===
  pin 1    PE2                  -> SPI4_SCK
  pin 4    PE5                  -> SPI4_MISO
  pin 5    PE6                  -> SPI4_MOSI
  pin 51   PA5                  -> SPI1_SCK
  pin 52   PA6                  -> SPI1_MISO
  pin 53   PA7                  -> SPI1_MOSI
  pin 139  PC10                 -> SPI5_SCK
  pin 140  PC11                 -> SPI5_MISO
  pin 141  PC12                 -> SPI5_MOSI
  pin 1    PE2                  -> SPI4_SCK
  pin 4    PE5                  -> SPI4_MISO
  pin 5    PE6                  -> SPI4_MOSI
  pin 51   PA5                  -> SPI1_SCK
  pin 52   PA6                  -> SPI1_MISO
  pin 53   PA7                  -> SPI1_MOSI
  pin 139  PC10                 -> SPI5_SCK
  pin 140  PC11                 -> SPI5_MISO
  pin 141  PC12                 -> SPI5_MOSI
